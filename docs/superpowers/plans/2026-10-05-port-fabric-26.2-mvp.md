# Port WorkerMod sang Fabric 26.2 — Implementation Plan (MVP)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Port mod Villager Workers (Forge 1.20.1) sang Fabric cho Minecraft 26.2 dưới dạng standalone, MVP chạy end-to-end với Farmer và Lumberjack.

**Architecture:** Mod standalone không phụ thuộc Recruits. Entity worker kế thừa `PathfinderMob` (vanilla). Work area là entity vô hình giữ AABB + state synced. AI dùng goal selector của vanilla. Network dùng Fabric API payload. Build qua GitHub Actions.

**Tech Stack:** Java 25, Fabric Loader 0.19.5, Fabric API 0.161.0+26.2, Loom 1.18-SNAPSHOT, Gradle 9.7.1, Mojang official mappings.

**Spec:** `docs/superpowers/specs/2026-10-05-port-fabric-26.2-design.md`

```java
package com.talhanation.workers.config;

import com.google.gson.Gson;
import com.google.gson.GsonBuilder;
import com.talhanation.workers.WorkersMain;
import net.fabricmc.loader.api.FabricLoader;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;

public class WorkersConfig {
	private static WorkersConfig instance;

	public int farmerCost = 10;
	public int lumberjackCost = 12;
	public int minerCost = 16;
	public int builderCost = 20;
	public int merchantCost = 30;
	public int courierCost = 20;
	public int cookCost = 25;
	public int animalPenMaxAnimals = 32;
	public boolean shouldOnlyPlacingBuildingsBePossible = false;
	public boolean shouldWorkAreaOnlyBeInFactionClaim = false;
	public boolean workersReplaceAllay = true;
	public BuildMode buildMode = BuildMode.FREE;
	public List<String> minerPickup = List.of("minecraft:torch", "minecraft:lantern", "minecraft:redstone");
	public List<String> minerIgnore = List.of("minecraft:air", "minecraft:cave_air", "minecraft:water");
	public List<String> farmerPickup = List.of("minecraft:wheat", "minecraft:wheat_seeds", "minecraft:carrot", "minecraft:potato");
	public List<String> animalFarmerPickup = List.of("minecraft:feather", "minecraft:leather", "minecraft:egg", "minecraft:beef");
	public List<String> lumberjackPickup = List.of("minecraft:stick", "minecraft:bee_nest");
	public List<String> fishermanPickup = List.of("minecraft:tropical_fish", "minecraft:pufferfish");

	// Trả về config đã load, load lần đầu nếu chưa có
	public static WorkersConfig get() {
		if (instance == null) load();
		return instance;
	}

	// Đọc config từ đĩa, tạo file default nếu chưa tồn tại
	public static void load() {
		Path path = FabricLoader.getInstance().getConfigDir().resolve("workers.json");
		Gson gson = new GsonBuilder().setPrettyPrinting().create();
		if (Files.exists(path)) {
			try {
				instance = gson.fromJson(Files.readString(path), WorkersConfig.class);
			} catch (IOException e) {
				WorkersMain.LOGGER.error("Không đọc được workers.json, dùng default", e);
				instance = new WorkersConfig();
			}
		} else {
			instance = new WorkersConfig();
			try {
				Files.writeString(path, gson.toJson(instance));
			} catch (IOException e) {
				WorkersMain.LOGGER.error("Không ghi được workers.json", e);
			}
		}
		if (instance == null) instance = new WorkersConfig();
	}
}
```

KHÔNG copy nguyên file reference (631 dòng) — bản đó kế thừa `AbstractChunkLoaderEntity` của Recruits. Viết lại với `PathfinderMob`, giữ đúng các API trong block Interfaces ở trên.

Khung chính:

```java
package com.talhanation.workers.entities;

import com.talhanation.workers.world.NeededItem;
import net.minecraft.core.BlockPos;
import net.minecraft.nbt.CompoundTag;
import net.minecraft.network.chat.Component;
import net.minecraft.network.syncher.EntityDataAccessor;
import net.minecraft.network.syncher.EntityDataSerializers;
import net.minecraft.network.syncher.SynchedEntityData;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.sounds.SoundEvents;
import net.minecraft.world.InteractionHand;
import net.minecraft.world.InteractionResult;
import net.minecraft.world.SimpleContainer;
import net.minecraft.world.entity.EntityType;
import net.minecraft.world.entity.PathfinderMob;
import net.minecraft.world.entity.ai.goal.Goal;
import net.minecraft.world.entity.item.ItemEntity;
import net.minecraft.world.entity.player.Player;
import net.minecraft.world.item.ItemStack;
import net.minecraft.world.level.Level;
import net.minecraft.world.level.block.Block;
import net.minecraft.world.level.block.Blocks;
import net.minecraft.world.level.block.state.BlockState;
import net.minecraft.world.level.storage.ValueInput;
import net.minecraft.world.level.storage.ValueOutput;

import javax.annotation.Nullable;
import java.util.ArrayList;
import java.util.List;
import java.util.Optional;
import java.util.UUID;
import java.util.function.Predicate;

public abstract class AbstractWorkerEntity extends PathfinderMob {
	public static final EntityDataAccessor<Optional<UUID>> OWNER_UUID =
			SynchedEntityData.defineId(AbstractWorkerEntity.class, EntityDataSerializers.OPTIONAL_UUID);
	public static final EntityDataAccessor<String> OWNER_NAME =
			SynchedEntityData.defineId(AbstractWorkerEntity.class, EntityDataSerializers.STRING);
	public static final EntityDataAccessor<Integer> FOLLOW_STATE =
			SynchedEntityData.defineId(AbstractWorkerEntity.class, EntityDataSerializers.INT);
	public static final EntityDataAccessor<Float> MORALE =
			SynchedEntityData.defineId(AbstractWorkerEntity.class, EntityDataSerializers.FLOAT);

	public static final int FOLLOW_STATE_WORK = 6;

	@Nullable public UUID homeAreaUUID = null;
	public boolean isFleeing = false;
	@Nullable public BlockPos fleeTarget = null;
	public List<NeededItem> neededItems = new ArrayList<>();
	public int farmedItems;
	public boolean forcedDeposit;
	@Nullable public UUID lastStorage;
	public final SimpleContainer inventory = new SimpleContainer(9);

	public AbstractWorkerEntity(EntityType<? extends AbstractWorkerEntity> entityType, Level level) {
		super(entityType, level);
	}

	// Đăng ký các goal chung cho mọi worker
	@Override
	protected void registerGoals() {
	}

	// Đọc dữ liệu owner và trạng thái từ NBT
	@Override
	protected void readAdditionalSaveData(ValueInput input) {
		super.readAdditionalSaveData(input);
		input.getString("ownerUUID").ifPresent(s -> setOwnerUUID(UUID.fromString(s)));
		homeAreaUUID = input.getString("homeAreaUUID").map(UUID::fromString).orElse(null);
		farmedItems = input.getIntOr("farmedItems", 0);
	}

	// Ghi dữ liệu owner và trạng thái xuống NBT
	@Override
	protected void addAdditionalSaveData(ValueOutput output) {
		super.addAdditionalSaveData(output);
		UUID owner = getOwnerUUID();
		if (owner != null) output.putString("ownerUUID", owner.toString());
		if (homeAreaUUID != null) output.putString("homeAreaUUID", homeAreaUUID.toString());
		output.putInt("farmedItems", farmedItems);
	}

	// Khai báo các trường dữ liệu đồng bộ client-server
	@Override
	protected void defineSynchedData(SynchedEntityData.Builder builder) {
		super.defineSynchedData(builder);
		builder.define(OWNER_UUID, Optional.empty());
		builder.define(OWNER_NAME, "");
		builder.define(FOLLOW_STATE, 0);
		builder.define(MORALE, 100.0f);
	}

	public boolean isOwned() { return getOwnerUUID() != null; }

	@Nullable
	public UUID getOwnerUUID() { return this.entityData.get(OWNER_UUID).orElse(null); }

	public String getOwnerName() { return this.entityData.get(OWNER_NAME); }

	public void setOwnerUUID(@Nullable UUID uuid) {
		this.entityData.set(OWNER_UUID, Optional.ofNullable(uuid));
	}

	public void setOwnerName(String name) { this.entityData.set(OWNER_NAME, name); }

	@Nullable
	public Player getOwner() {
		UUID uuid = getOwnerUUID();
		if (uuid == null) return null;
		return level().getPlayerByUUID(uuid);
	}

	public void setOwner(Player player) {
		setOwnerUUID(player.getUUID());
		setOwnerName(player.getName().getString());
	}

	public int getFollowState() { return this.entityData.get(FOLLOW_STATE); }

	public void setFollowState(int state) { this.entityData.set(FOLLOW_STATE, state); }

	public float getMorale() { return this.entityData.get(MORALE); }

	public void setMorale(float value) { this.entityData.set(MORALE, value); }

	public boolean isWorking() { return getFollowState() == FOLLOW_STATE_WORK; }

	public SimpleContainer getInventory() { return this.inventory; }

	// Đếm số item khớp điều kiện trong inventory
	public int countMatchingItems(Predicate<ItemStack> predicate) {
		int count = 0;
		for (int i = 0; i < inventory.getContainerSize(); i++) {
			ItemStack stack = inventory.getItem(i);
			if (!stack.isEmpty() && predicate.test(stack)) count += stack.getCount();
		}
		return count;
	}

	// Đếm số slot chứa item khớp điều kiện
	public int countMatchingStacks(Predicate<ItemStack> predicate) {
		int count = 0;
		for (int i = 0; i < inventory.getContainerSize(); i++) {
			ItemStack stack = inventory.getItem(i);
			if (!stack.isEmpty() && predicate.test(stack)) count++;
		}
		return count;
	}

	// Kiểm tra inventory còn chỗ chứa item
	public boolean canAddItem(ItemStack stack) {
		for (int i = 0; i < inventory.getContainerSize(); i++) {
			ItemStack slot = inventory.getItem(i);
			if (slot.isEmpty()) return true;
			if (ItemStack.isSameItemSameComponents(slot, stack)
					&& slot.getCount() + stack.getCount() <= slot.getMaxStackSize()) return true;
		}
		return false;
	}

	// Thêm item vào inventory, trả về phần còn dư
	public ItemStack addItem(ItemStack stack) {
		ItemStack remainder = stack.copy();
		for (int i = 0; i < inventory.getContainerSize() && !remainder.isEmpty(); i++) {
			ItemStack slot = inventory.getItem(i);
			if (slot.isEmpty()) {
				inventory.setItem(i, remainder);
				return ItemStack.EMPTY;
			}
			if (ItemStack.isSameItemSameComponents(slot, remainder)) {
				int space = slot.getMaxStackSize() - slot.getCount();
				int moved = Math.min(space, remainder.getCount());
				slot.grow(moved);
				remainder.shrink(moved);
			}
		}
		return remainder;
	}

	// Trừ độ bền item đang cầm trên tay chính
	public void damageMainHandItem() {
		ItemStack stack = getMainHandItem();
		if (!stack.isEmpty() && stack.isDamageableItem()) {
			stack.hurtAndBreak(1, this, net.minecraft.world.entity.EquipmentSlot.MAINHAND);
		}
	}

	public boolean wantsToPickUp(ItemStack stack) { return false; }

	public boolean wantsToKeep(ItemStack stack) { return false; }

	public abstract Goal getWorkGoal();

	// Xử lý tương tác chuột phải: hire nếu chưa có chủ, mở GUI nếu là chủ
	@Override
	protected InteractionResult mobInteract(Player player, InteractionHand hand) {
		if (!level().isClientSide()) {
			if (!isOwned()) {
				hire(player);
			}
		}
		return InteractionResult.SUCCESS;
	}

	// Gán chủ cho worker
	public void hire(Player player) {
		setOwner(player);
		playSound(SoundEvents.VILLAGER_YES, 1.0f, 1.0f);
	}

	// Giải phóng worker khỏi chủ
	public void disband() {
		setOwnerUUID(null);
		setOwnerName("");
		setFollowState(0);
	}
}

import net.minecraft.network.protocol.common.custom.CustomPacketPayload;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.entity.Entity;

public record AddWorkAreaPayload(BlockPos pos, int typeIndex) implements CustomPacketPayload {
	public static final CustomPacketPayload.Type<AddWorkAreaPayload> TYPE =
			new CustomPacketPayload.Type<>(WorkersMain.id("add_work_area"));

	public static final StreamCodec<RegistryFriendlyByteBuf, AddWorkAreaPayload> CODEC =
			StreamCodec.composite(
					BlockPos.STREAM_CODEC, AddWorkAreaPayload::pos,
					net.minecraft.network.codec.ByteBufCodecs.VAR_INT, AddWorkAreaPayload::typeIndex,
					AddWorkAreaPayload::new);

	@Override
	public Type<? extends CustomPacketPayload> type() { return TYPE; }

	// Tạo work area mới theo loại được chọn
	public static void handle(AddWorkAreaPayload payload, ServerPlayer player) {
		BlockPos pos = payload.pos();
		WorkAreaTypes type = WorkAreaTypes.fromIndex(payload.typeIndex());

		if (!player.level().isEmptyBlock(pos.above())) {
			player.sendSystemMessage(Component.translatable("gui.workers.area.no_air_above"));
			return;
		}
		if (WorkersConfig.get().shouldOnlyPlacingBuildingsBePossible && type != WorkAreaTypes.BUILDING) {
			return;
		}

		AbstractWorkAreaEntity area = switch (type) {
			case CROPAREA -> new CropArea(ModEntityTypes.CROPAREA, player.level());
			case LUMBER -> new LumberArea(ModEntityTypes.LUMBERAREA, player.level());
			case STORAGE -> new StorageArea(ModEntityTypes.STORAGEAREA, player.level());
			default -> null;
		};
		if (area == null) return;

		area.setFacing(player.getDirection());
		area.moveTo(pos.above(), 0, 0);
		area.setPlayerName(player.getName().getString());
		area.setPlayerUUID(player.getUUID());
		area.setTeamStringID(player.getTeam() != null ? player.getTeam().getName() : "");

		if (AbstractWorkAreaEntity.isAreaOverlapping(player.level(), null, area.getArea())) {
			player.sendSystemMessage(Component.translatable("gui.workers.area.overlapping"));
			return;
		}

		player.level().addFreshEntity(area);
	}
}
```

package com.talhanation.workers.client;

import com.talhanation.workers.client.gui.CropAreaScreen;
import com.talhanation.workers.client.gui.LumberAreaScreen;
import com.talhanation.workers.client.gui.StorageAreaScreen;
import com.talhanation.workers.entities.workarea.AbstractWorkAreaEntity;
import com.talhanation.workers.entities.workarea.CropArea;
import com.talhanation.workers.entities.workarea.LumberArea;
import com.talhanation.workers.entities.workarea.StorageArea;
import com.talhanation.workers.network.OpenWorkAreaScreenPayload;
import net.fabricmc.fabric.api.client.networking.v1.ClientPlayNetworking;
import net.minecraft.client.Minecraft;

public class WorkersClientNetwork {
	// Đăng ký nhận payload mở màn hình work area
	public static void register() {
		ClientPlayNetworking.registerGlobalReceiver(OpenWorkAreaScreenPayload.TYPE, (payload, context) -> {
			Minecraft client = context.client();
			client.execute(() -> {
				if (client.player == null) return;
				client.player.level().getEntitiesOfClass(AbstractWorkAreaEntity.class,
								client.player.getBoundingBox().inflate(16.0D),
								area -> area.getUUID().equals(payload.areaUUID()))
						.stream().findAny().ifPresent(area -> {
							if (area instanceof CropArea crop) client.setScreen(new CropAreaScreen(crop));
							else if (area instanceof LumberArea lumber) client.setScreen(new LumberAreaScreen(lumber));
							else if (area instanceof StorageArea storage) client.setScreen(new StorageAreaScreen(storage));
						});
			});
		});
	}
}

package com.talhanation.workers.items;

import com.talhanation.workers.WorkAreaTypes;

public class WorkAreaToolItem extends Item {
